calculate_mtrspony_pop_count_equal = {
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { is_same_species = owner_main_species }
		}
		variable = equal_pops
	}
	divide_variable = { which = equal_pops value = 9 }
	round_variable_to_closest = { which = equal_pops value = 5 }

	set_variable = { which = pony_pops value = equal_pops }
	set_variable = { which = griffon_pops value = equal_pops }
	set_variable = { which = zebra_pops value = equal_pops }
	set_variable = { which = deer_pops value = equal_pops }
	set_variable = { which = changeling_pops value = equal_pops }
	set_variable = { which = hippogriff_pops value = equal_pops }
	set_variable = { which = kirin_pops value = equal_pops }
	set_variable = { which = yak_pops value = equal_pops }
	set_variable = { which = dragon_pops value = equal_pops }

	calculate_pops_to_remove = yes
}

calculate_pops_to_remove = {
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { is_same_species = owner_main_species }
		}
		variable = pops_to_remove
	}
	if = {
		limit = { owner = { has_mtrspony_pony_portrait = yes } }

		subtract_variable = { which = pops_to_remove value = pony_pops }
	}
	else_if = {
		limit = { owner = { species_portrait = mtrspony_griffon } }

		subtract_variable = { which = pops_to_remove value = griffon_pops }
	}
	else_if = {

				add_trait_to_mtrspony_species_effect = { SPECIES = event_target:changeling_species TRAIT = trait_incubator }
			}
			else_if = {
				limit = { has_biogenesis_dlc = yes }

				add_trait_to_mtrspony_species_effect = { SPECIES = event_target:changeling_species TRAIT = trait_flight }
			}
			else = {
				remove_trait_from_mtrspony_species_effect = { SPECIES = event_target:changeling_species TRAIT = trait_wasteful }
				remove_trait_from_mtrspony_species_effect = { SPECIES = event_target:changeling_species TRAIT = trait_solitary }
				add_trait_to_mtrspony_species_effect = { SPECIES = event_target:changeling_species TRAIT = trait_enduring }
			}
		}
		if = {
			limit = { NOT = { owner_main_species = { is_same_species = event_target:hippogriff_species } } }

			if = {
				limit = { has_biogenesis_dlc = yes }

				add_trait_to_mtrspony_species_effect = { SPECIES = event_target:hippogriff_species TRAIT = trait_flight }
			}
			else = {
				add_trait_to_mtrspony_species_effect = { SPECIES = event_target:hippogriff_species TRAIT = trait_charismatic }
			}
		}
		if = {
			limit = { NOT = { owner_main_species = { is_same_species = event_target:kirin_species } } }

			add_trait_to_mtrspony_species_effect = { SPECIES = event_target:kirin_species TRAIT = trait_resilient }
			add_trait_to_mtrspony_species_effect = { SPECIES = event_target:kirin_species TRAIT = trait_strong }
		}
		if = {
			limit = { NOT = { owner_main_species = { is_same_species = event_target:yak_species } } }

			add_trait_to_mtrspony_species_effect = { SPECIES = event_target:yak_species TRAIT = trait_resilient }
			add_trait_to_mtrspony_species_effect = { SPECIES = event_target:yak_species TRAIT = trait_enduring }
		}
		if = {
			limit = { NOT = { owner_main_species = { is_same_species = event_target:dragon_species } } }

			replace_traits_on_mtrspony_species_effect = { SPECIES = event_target:dragon_species REMOVE_TRAIT = trait_strong ADD_TRAIT = trait_very_strong }
		}
	}
	else = { } # TODO: REVERT TRAIT CHANGES

	# Other
	if = {
		limit = { owner_main_species = { has_trait = trait_aquatic } }

		# TODO: This eats up 2 trait points, rest of the traits will need adjustments.
		add_trait_to_mtrspony_species_effect = { TRAIT = trait_aquatic }
	}
	else = { remove_trait_from_mtrspony_species_effect = { TRAIT = trait_aquatic } }

	[[!IGNORE_PC] add_original_species_pc_trait = yes ]
}

add_trait_to_mtrspony_species_effect = {
	every_galaxy_species = {
		limit = {
			[[SPECIES] is_same_species = $SPECIES$ ]
			[[!SPECIES] is_exact_same_species_with_empire_faustian_races = { COUNTRY = PREVPREV } ]
			NOT = { has_trait = $TRAIT$ }
		}

		if = {
			limit = { has_species_flag = species_creator@PREV.owner }

			change_species_characteristics = { add_trait = $TRAIT$ }
		}
		else = {
			modify_species = {
				species = THIS
				add_trait = $TRAIT$

					save_event_target_for_modified_species = yes
				}
			}
		}
	}
}

	}

add_original_species_pc_trait_effect = {

		limit = {
			[[SPECIES] is_same_species = $SPECIES$ ]
			[[!SPECIES] is_exact_same_species_with_empire_faustian_races = { COUNTRY = PREV.owner } ]
			NOT = { has_trait = $TRAIT$ }
		}

		if = {
			limit = { has_species_flag = species_creator@PREV.owner }

			set_habitability_trait = { trait = $TRAIT$ }
		}
		else = {
			modify_species = {
				species = THIS
				add_trait = $TRAIT$

				effect = {
					set_habitability_trait = { trait = $TRAIT$ } # Actually removes the other habitability traits.
					set_species_homeworld = PREVPREV.owner.capital_scope

					# Game doesn't like too much scripted effects within scripted effects.
					if = {
						limit = { PREV = { is_same_species = event_target:pony_species } }

						PREVPREV.owner = { PREV = { save_global_event_target_as = pony_species@PREV } }
					}
					else_if = {
						limit = { PREV = { is_same_species = event_target:griffon_species } }

						PREVPREV.owner = { PREV = { save_global_event_target_as = griffon_species@PREV } }

remove_x_ROOT_species_pops = {
	set_variable = { which = pops_to_remove_div_5 value = pops_to_remove }
	divide_variable = { which = pops_to_remove_div_5 value = 5 }

	while = {
		count = pops_to_remove_div_5

		random_owned_pop_group = {
			limit = { is_same_species = owner.species }

			kill_pop_group = {
				pop_group = this
				amount = 5
			}
		}
	}
}

# Duplicate of "ethic_leader_creator" where species is a variable, rather than being crated during the event.
mtrspony_ethic_leader_creator = {
	create_country = {
		name = "NAME_leader_country"
		type = faction
		auto_delete = yes
		flag = {
			icon = {
				category = "special"
				file = "unknown.dds"
			}
			background = {
				category = "backgrounds"
				file = "00_solid.dds"
			}
			colors = {
				"red"
				"red"
				"null"
				"null"
			}
		}
		effect = {
			save_event_target_as = leader_country
		}
	}
	event_target:leader_country = {
		$SPECIES$ = {
			save_event_target_as = leader_species
		}
		create_leader = {
			class = $CLASS$
			tier = $TIER$
			species = event_target:leader_species
			name = $NAME$
			skill = $LEVEL$
			gender = $GENDER$
			event_leader = no # Leader is allowed to lead factions and win elections
			background_ethic = ethic_$ETHIC$
			skip_background_generation = yes
			custom_description = $DESC$
			custom_catch_phrase = $CATCH_PHRASE$
			immortal = $IMMORTAL$
			hide_age = $HIDEAGE|IMMORTAL$ # Matches HIDEAGE, Fallbacks on IMMORTAL

			randomize_traits = no
			effect = {
				set_skill = $LEVEL$
				save_event_target_as = ethic_leader
				set_leader_flag = is_in_recruit_window
				set_variable = { which = leader_pending_negative_traits_unmodified value = 1 }
				change_leader_portrait = $LEADER_PORTRAIT$
			}
		}
	}
	event_target:ethic_leader = {
		set_owner = root
		# This is a hack to make the leader not be added to player country.
		# We must call ethic_leader_hire on the hire event
		# We must call ethic_leader_kill on the dismiss event
		exile_leader_as = ethic_leader_exile
	}
}
