save_current_mtrspony_species = {
	every_country = {
		if = {
			limit = { has_mtrspony_pony_portrait = yes }

			owner_main_species = {
				save_global_event_target_as = pony_species
				set_global_flag = pony_species_created
				set_species_flag = species_creator@PREV
			}
		}
		else_if = {
			limit = { species_portrait = mtrspony_griffon }

			owner_main_species = {
				save_global_event_target_as = griffon_species
				set_global_flag = griffon_species_created
				set_species_flag = species_creator@PREV
			}
		}
		else_if = {
			limit = { species_portrait = mtrspony_zebra }

			owner_main_species = {
				save_global_event_target_as = zebra_species
				set_global_flag = zebra_species_created

			limit = { species_portrait = mtrspony_changeling }

			owner_main_species = {
				save_global_event_target_as = changeling_species
				set_global_flag = changeling_species_created
				set_species_flag = species_creator@PREV
			}
		}
		else_if = {
			limit = { species_portrait = mtrspony_hippogriff }

			owner_main_species = {
				save_global_event_target_as = hippogriff_species
				set_global_flag = hippogriff_species_created
				set_species_flag = species_creator@PREV
			}

		else_if = {
			limit = { species_portrait = mtrspony_kirin }

				save_global_event_target_as = kirin_species
				set_global_flag = kirin_species_created

add_trait_to_mtrspony_species_effect = {
	every_owned_pop_group = {
		limit = {
			[[SPECIES] is_same_species = $SPECIES$ ]
			[[!SPECIES] is_same_species_with_other_faustian_races = yes	]
			NOT = { has_trait = $TRAIT$ }
		}

		if = {
			limit = { species = { has_species_flag = species_creator@PREV.owner } }

			species = { change_species_characteristics = { add_trait = $TRAIT$ } }
		}
		else = {
			modify_species = {
				species = species
				add_trait = $TRAIT$
			}
		}
	}
}

			}
		}
	}
}

add_original_species_pc_trait_effect = {
	every_owned_pop_group = {
		limit = {
			[[SPECIES] is_same_species = $SPECIES$ ]
			[[!SPECIES] is_same_species_with_other_faustian_races = yes	]
			NOT = { has_trait = $TRAIT$ }
		}

		if = {
			limit = { species = { has_species_flag = species_creator@PREV.owner } }

			species = { set_habitability_trait = { trait = $TRAIT$ } }
		}
		else = {
			modify_species = {
				species = species
				add_trait = $TRAIT$

				effect = {
					set_habitability_trait = { trait = $TRAIT$ } # Actually removes the other habitability traits.
					set_species_homeworld = PREV.owner.capital_scope
				}
			}
		}
	}
}

remove_x_ROOT_species_pops = {
	set_variable = { which = pops_to_remove_div_5 value = pops_to_remove }
	divide_variable = { which = pops_to_remove_div_5 value = 5 }

	while = {
		count = pops_to_remove_div_5

		random_owned_pop_group = {
			limit = { is_same_species = owner.species }

			kill_pop_group = {
				pop_group = this
				amount = 5
			}
		}
	}
}

generate_faust_deposits_and_blockers = {
	optimize_memory
	clear_deposits = yes
	add_deposit = d_magical_jungles
	add_deposit = d_zebharan_solar_farms
	add_deposit = d_thundarian_range
	add_deposit = d_griffking_dam
	add_deposit = d_rock_farms
	add_deposit = d_andelayan_mines
	add_deposit = d_panthalassic_ore_veins
	add_deposit = d_evi_valley_farmlands
	add_deposit = d_seaquestrian_algae_farms
	add_deposit = d_neighle_river_delta
	add_deposit = d_crystal_mountains
	if = {
		limit = {
			owner = { is_machine_empire = yes }
		}
		add_deposit = d_rushing_waterfalls
	}

	if = {
		limit = {
			owner = {
				is_environmental_architects_empire = yes
			}
		}
		add_random_natural_blocker_and_give_tech = yes

		if = {
			limit = {
				owner = { has_valid_civic = civic_machine_servitor }
			}
			add_deposit = d_unsupervised_settlement
		}
	}
	else = {
		add_deposit = d_failing_infrastructure_faust
		if = {
			limit = {
				owner = { has_valid_civic = civic_machine_servitor }
			}
			add_deposit = d_hot_springs
			add_deposit = d_strip_mine
			add_deposit = d_unsupervised_settlement
		}
		else_if = {
			limit = {
				owner = {
					is_hive_empire = yes
				}
			}

			add_deposit = d_collapsed_burrows
		}
		else = {
			add_deposit = d_decrepit_dwellings
		}
	}

	if = {
		limit = {
			owner = {
				is_lithoid_empire = yes
				NOT = { has_valid_civic = civic_hive_devouring_swarm }
			}
		}
		while = {
			count = 4
			add_deposit = d_hibernating_lithoids
		}
	}

	if = {
		limit = {
			owner = {
				is_lithoid_devouring_swarm = yes
			}
		}
		while = {
			count = 2
			add_deposit = d_hibernating_lithoids
		}
	}

	if = {
		limit = {
			owner = {
				has_origin = origin_life_seeded
			}
		}

		set_planet_size = 30

		add_deposit = d_dust_caverns
		add_deposit = d_bubbling_swamp
		add_deposit = d_crystalline_caverns
	}

	if = {
		limit = {
			is_planet_class = pc_desert # Just in case players cause trouble
		}
		set_planet_flag = tasty_desert
	}
}
