save_current_mtrspony_species = {
	if = {
		limit = { has_mtrspony_pony_portrait = yes }

		owner_main_species = { save_global_event_target_as = pony_species }
	}
	else_if = {
		limit = { species_portrait = mtrspony_griffon }

		owner_main_species = { save_global_event_target_as = griffon_species }

	else_if = {

	else_if = {

		limit = { owner = { has_mtrspony_pony_portrait = yes } }

		subtract_variable = { which = pops_to_remove value = pony_pops }
	}
	else_if = {
		limit = { owner = { species_portrait = mtrspony_griffon } }

		subtract_variable = { which = pops_to_remove value = griffon_pops }
	}
	else_if = {
		limit = { owner = { species_portrait = mtrspony_hippogriff } }
