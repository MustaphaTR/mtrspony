namespace = mtrspony_setup

event = {
	id = mtrspony_setup.1

	hide_window = yes
	is_triggered_only = yes

	immediate = {
		save_current_mtrspony_species = yes

		every_playable_country = {
			limit = {
				has_mtrspony_portrait = yes
				is_mtrspony_prescripted_country = no
			}

			create_other_mtrspony_species = yes
			every_owned_planet = { set_mtrspony_setup_variables = yes }

			if = {
				limit = { is_ai = no }

				every_owned_planet = { set_planet_flag = mtrspony_planet_to_setup@PREV }
				set_country_flag = mtrspony_country_to_setup@THIS
				set_country_flag = should_get_setup_event
			}
			else = {
				random_country = {
					limit = { is_ai = no }

					PREV = {
						every_owned_planet = { set_planet_flag = mtrspony_planet_to_setup@PREVPREV }
						set_country_flag = mtrspony_country_to_setup@PREV
					}
					set_country_flag = should_get_setup_event
				}
			}
		}
		every_playable_country = {
			limit = { has_country_flag = should_get_setup_event }

			if = {
				limit = {
					has_mtrspony_portrait = yes
					is_mtrspony_prescripted_country = no
				}

				capital_scope = { save_global_event_target_as = mtrspony_species_setup_current_planet@PREV }

				country_event = { id = mtrspony_setup.10 scopes = { from = capital_scope } }
			}
			else = {
				random_playable_country = {
					limit = { has_country_flag = mtrspony_country_to_setup@PREV }

					capital_scope = { save_global_event_target_as = mtrspony_species_setup_current_planet@PREVPREV }
					PREV = { country_event = { id = mtrspony_setup.10 scopes = { from = PREV.capital_scope } } }
				}
			}
		}
	}
}
