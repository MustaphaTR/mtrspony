namespace = mtrspony_setup

country_event = {
	id = mtrspony_setup.1
	title = "mtrspony_setup.1.name"
	desc = "mtrspony_setup.1.desc"

	picture = GFX_evt_voting

	is_triggered_only = yes

	trigger = {
		is_ai = no
		has_mtrspony_portrait = yes
		is_mtrspony_prescripted_country = no
	}

	immediate = {
		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		capital_scope = { set_mtrspony_setup_variables = yes }
	}

	option = {
		name = mtrspony_setup.1.a
	}
	option = {
		name = mtrspony_setup.1.b

		hidden_effect = {
			capital_scope = {
				set_variable = { which = pony_pops value = 100 }
				set_variable = { which = griffon_pops value = 100 }
				set_variable = { which = hippogriff_pops value = 100 }
				set_variable = { which = dragon_pops value = 100 }
				set_variable = { which = kirin_pops value = 100 }
				set_variable = { which = changeling_pops value = 100 }
				set_variable = { which = yak_pops value = 100 }
				set_variable = { which = zebra_pops value = 100 }
				set_variable = { which = pops_to_remove value = 700 }
				spawn_other_mtrspony_species = yes
			}

			set_country_flag = using_mtrspony_multi_species_setup
		}
	}
	option = {
		name = mtrspony_setup.1.c
		hidden_effect = {
			capital_scope = {
				calculate_mtrspony_pop_count_default = yes
				spawn_other_mtrspony_species = yes
			}

			set_country_flag = using_mtrspony_multi_species_setup
		}
	}
	option = {
		name = mtrspony_setup.1.d
		hidden_effect = {
			capital_scope = {
				calculate_mtrspony_pop_count_keep_majority = yes
				spawn_other_mtrspony_species = yes
			}

			set_country_flag = using_mtrspony_multi_species_setup
		}
	}
}

# Send a setup event to a player country about the AI countries as well.
country_event = {
	id = mtrspony_setup.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		has_mtrspony_portrait = yes
		is_mtrspony_prescripted_country = no
	}

	immediate = {
		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		capital_scope = { set_mtrspony_setup_variables = yes }

		random_country = {
			limit = { is_ai = no }

			country_event = { id = mtrspony_setup.3 }
		}
	}
}
