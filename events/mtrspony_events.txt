namespace = mtrspony

# After taking Psionics tradition give Latent Psionics trait to other Faustian species too.
country_event = {
	id = mtrspony.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		using_mtrspony_multi_species_setup = yes
		has_tradition = tr_psionics_adopt
		NOT = { has_country_flag = applied_multi_species_psionics_workaround_1 }
	}
	immediate = {
		set_country_flag = applied_multi_species_psionics_workaround_1

		every_owned_pop_group = {
			limit = {
				is_same_species_with_other_faustian_races = yes
				species = {
					has_psionic_species_trait = no
				}
			}
			modify_species = {
				species = this
				add_trait = trait_latent_psionic
			}
		}
	}
}

# After taking Great Awakening tradition give Psionics trait to other Faustian species too.
country_event = {
	id = mtrspony.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		using_mtrspony_multi_species_setup = yes
		has_tradition = tr_psionics_psionic_assimilation
		NOT = { has_country_flag = applied_multi_species_psionics_workaround_2 }
	}
	immediate = {
		set_country_flag = applied_multi_species_psionics_workaround_2

		every_owned_pop_group = {
			limit = {
				is_same_species_with_other_faustian_races = yes
				has_trait = trait_latent_psionic
			}
			modify_species = {
				species = this
				remove_trait = trait_latent_psionic
				add_trait = trait_psionic
			}
		}
		every_country = {
			limit = {
				OR = {
					is_country_type = default
					is_country_type = fallen_empire
					is_country_type = awakened_fallen_empire
				}
			}
			every_owned_leader = {
				limit = {
					is_same_species_with_other_faustian_races = yes
					NOR = {
						has_trait = leader_trait_psionic
						has_leader_flag = renowned_leader
						has_leader_flag = legendary_leader
					}
				}
				add_trait = {
					trait = leader_trait_psionic
					show_message = no
				}
				owner = {
					if = {
						limit = {
							NOT = {
								has_country_flag = psionic_leader_toast
							}
						}
						set_timed_country_flag = {
							flag = psionic_leader_toast
							days = 30
						}
					}
				}
			}

			every_pool_leader = {
				limit = {
					is_same_species_with_other_faustian_races = yes
					NOT = { has_trait = leader_trait_psionic }
				}
				add_trait = {
					trait = leader_trait_psionic
					show_message = no
				}
				owner = {
					if = {
						limit = {
							NOT = {
								has_country_flag = psionic_leader_toast
							}
						}
						set_timed_country_flag = {
							flag = psionic_leader_toast
							days = 30
						}
					}
				}
			}

			if = {
				limit = {
					has_country_flag = psionic_leader_toast
				}
				create_message = {
					type = MESSAGE_ALL_GAINED_TRAIT
					localization = MESSAGE_ALL_GAINED_TRAIT_DESC
					days = @toast_message_days
					variable = {
						type = key
						value = leader_trait_psionic
						localization = TRAIT
					}
					variable = {
						key = TRAIT_KEY
						value = leader_trait_psionic
					}
					variable = {
						key = "border"
						value = "GFX_invisible" #TODO Common/ no rarity frame
					}
				}
			}
		}
	}
}

# Age up immortral leaders.
country_event = {
	id = mtrspony.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		leader = { has_trait = leader_trait_mtrspony_immortal }
	}
	immediate = {
		leader = {
			export_trigger_value_to_variable = {
				trigger = leader_age
				variable = age
			}
			set_variable_to_random_value = {
				which = extra_age
				min = 200
				max = 2000
				rounded = yes
			}
			change_variable = {
				which = age
				value = extra_age
			}
			set_age = age
		}
	}
}

# After taking Enviromental Integration tradition give Mutagenic Habitability trait to other Faustian species too.
# Normally, there's a Council Agenda to apply this to other species instantly, but Evolutionary Predetors do not get it. So better do it automatically.
country_event = {
	id = mtrspony.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		using_mtrspony_multi_species_setup = yes
		has_tradition = tr_mutation_adopt
		has_mutation_environmental_integration_tradition = yes
		NOT = { has_country_flag = applied_multi_species_mutation_workaround_1 }
	}
	immediate = {
		set_country_flag = applied_multi_species_mutation_workaround_1

		every_owned_pop_group = {
			limit = {
				is_same_species_with_other_faustian_races = yes
				species = {
					NOT = {
						has_trait = trait_auto_hab_preference
					}
					OR = {
						is_archetype = BIOLOGICAL
						is_archetype = LITHOID
					}
				}
			}
			species = { set_habitability_trait = { trait = trait_auto_hab_preference } }
		}
	}
}
