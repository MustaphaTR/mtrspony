# After taking Psionics tradition give Latent Psionics trait to other Faustian species too.
country_event = {
	id = mtrspony.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		using_mtrspony_multi_species_setup = yes
		OR = {
			has_tradition = tr_psionics_adopt
			has_tradition = tr_psionics_shroud_adopt
		}
		NOT = { has_country_flag = applied_multi_species_psionics_workaround_1 }
	}
	immediate = {
		set_country_flag = applied_multi_species_psionics_workaround_1

		every_owned_species = {
			limit = {
				NOT = { is_same_species = ROOT }
				is_same_species_with_faustian_races = yes
				has_psionic_species_trait = no
			}
			save_event_target_as = changing_species
			add_latent_psionic_trait = yes
			last_created_species = { save_event_target_as = psionic_species }

			ROOT = {
				every_owned_pop_group = {
					limit = { is_exact_same_species = event_target:changing_species }
					change_species = event_target:psionic_species
				}
				every_owned_leader = {
					limit = { is_exact_same_species = event_target:changing_species }
					change_species = event_target:psionic_species
				}
				every_pool_leader = {
					limit = { is_exact_same_species = event_target:changing_species }
					change_species = event_target:psionic_species
				}
				every_envoy = {
					limit = { is_exact_same_species = event_target:changing_species }
					change_species = event_target:psionic_species
				}
				every_owned_army = {
					limit = {
						exists = species
						is_exact_same_species = event_target:changing_species
					}
					change_species = event_target:psionic_species
				}
				every_controlled_ship = {
					limit = {
						is_ship_class = shipclass_colonizer
						is_exact_same_species = event_target:changing_species
					}		
					change_species = event_target:psionic_species
				}
			}
		}
	}
}

# Age up immortral leaders.
country_event = {
	id = mtrspony.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		leader = { has_trait = leader_trait_mtrspony_immortal }
	}
	immediate = {
		leader = {
			export_trigger_value_to_variable = {
				trigger = leader_age
				variable = age
			}
			set_variable_to_random_value = {
				which = extra_age
				min = 200
				max = 2000
				rounded = yes
			}
			change_variable = {
				which = age
				value = extra_age
			}
			set_age = age
		}
	}
}

# After taking Enviromental Integration tradition give Mutagenic Habitability trait to other Faustian species too.
# Normally, there's a Council Agenda to apply this to other species instantly, but Evolutionary Predetors do not get it. So better do it automatically.
country_event = {
	id = mtrspony.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		using_mtrspony_multi_species_setup = yes
		has_tradition = tr_mutation_adopt
		has_mutation_environmental_integration_tradition = yes
		NOT = { has_country_flag = applied_multi_species_mutation_workaround_1 }
	}
	immediate = {
		set_country_flag = applied_multi_species_mutation_workaround_1

		every_owned_pop_group = {
			limit = {
				is_same_species_with_other_faustian_races = yes
				species = {
					NOT = {
						has_trait = trait_auto_hab_preference
					}
					OR = {
						is_archetype = BIOLOGICAL
						is_archetype = LITHOID
					}
				}
			}
			species = { set_habitability_trait = { trait = trait_auto_hab_preference } }
		}
	}
}

			}

		}
	}
}

			if = {
				limit = { has_trait = leader_trait_paranoid_military }

# Handle cutie mark traits of existing leaders when species are modified.
# This = Country
# From = Species ( Post Modification )
# FromFrom = Species ( Original )
# Prev = Planet
country_event = {
	id = mtrspony.8
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		from = {
			save_event_target_as = new_species
		}
		every_owned_leader = {
			limit = {
				is_exact_same_species = event_target:new_species
			}
			if = {
				limit = {
					species = { has_trait = trait_cutie_marks }
					has_cutie_mark_trait = no
					is_event_leader = no # To disable nodes from getting cutie marks, hopefully doesn't disable anything else.
				}
				add_named_leader_cutie_mark_traits = yes
				if = {
					limit = { has_cutie_mark_trait = no }
					give_random_cutie_mark_trait = yes
				}
			}
			else_if = {
				limit = {
					species = { NOT = { has_trait = trait_cutie_marks } }
					has_cutie_mark_trait = yes
				}
				remove_cutie_mark_traits = yes
			}
		}

		every_pool_leader = {
			limit = {
				is_exact_same_species = event_target:new_species
			}
			if = {
				limit = {
					species = { has_trait = trait_cutie_marks }
					has_cutie_mark_trait = no
				}
				give_random_cutie_mark_trait = yes
			}
			else_if = {
				limit = {
					species = { NOT = { has_trait = trait_cutie_marks } }
					has_cutie_mark_trait = yes
				}
				remove_cutie_mark_traits = yes
			}
		}
	}
}
