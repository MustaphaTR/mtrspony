namespace = gestart

#Setup Convocation of Creatures
country_event = {
	id = gestart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = griffonian_empire
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_grover_xi"
			custom_description = "PRESCRIPTED_ruler_name_grover_xi_desc"
			species = ROOT.species
			gender = male
			class = official
			skill = 1
			set_age = 19
			background_ethic = "ethic_authoritarian"
			randomize_traits = no
			traits = {
				trait = trait_ruler_charismatic
			}

			effect = {
				save_global_event_target_as = leader_grover_xi
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_grover_xi_predefined"
				ROOT = { assign_leader = event_target:leader_grover_xi }
			}
		}

		create_pony_species = yes
		event_target:pony_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_hippogriff_species = yes
		event_target:hippogriff_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_dragon_species = yes
		event_target:dragon_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_kirin_species = yes
		event_target:kirin_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_changeling_species = yes
		event_target:changeling_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_zebra_species = yes
		event_target:zebra_species = {
			set_citizenship_type = { country = ROOT type = citizenship_limited }
			# set_military_service_type = { country = ROOT type = military_service_full }
		}

		capital_scope = {
			while = {
				count = 17

				random_owned_pop = { kill_pop = yes }
			}
			while = {
				count = 7

				create_pop = {
					species = event_target:pony_species
					ethos = random
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:hippogriff_species
					ethos = random
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:dragon_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:kirin_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:changeling_species
					ethos = random
				}
			}
			while = {
				count = 4

				create_pop = {
					species = event_target:zebra_species
					ethos = random
				}
			}
		}

		# Remove the auto-generated Grover XI.
		random_owned_leader = {
			limit = {
				NOT = { has_leader_flag = is_created_leader }
				has_trait = trait_ruler_charismatic
			}

			kill_leader = {
				show_notification = no
				fire = yes
			}
		}

		event_target:leader_grover_xi = {
			set_name = "PRESCRIPTED_ruler_name_grover_xi"
		}
	}
}

# Reapply named character potraits, they seem to reset after loading a save or when species got modified.

	}

		}
	}
