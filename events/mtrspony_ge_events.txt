namespace = gestart

#Setup Convocation of Creatures
country_event = {
	id = gestart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = griffonian_empire
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_grover_xi"
			custom_description = "PRESCRIPTED_ruler_name_grover_xi_desc"
			species = ROOT.species
			gender = male
			class = governor
			skill = 1
			set_age = 19
			background_ethic = "ethic_authoritarian"
			randomize_traits = no
			traits = {
				trait = trait_ruler_charismatic
			}

			effect = {
				save_event_target_as = leader_grover_xi
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_grover_xi_predefined"
				ROOT = { assign_leader = event_target:leader_grover_xi }
			}
		}

		create_species = {
			name = "PRESCRIPTED_species_pony"
			plural = "PRESCRIPTED_species_plural_pony"
			adjective = "PRESCRIPTED_species_adjective_pony"
			class = MTRSPONY
			portrait = mtrspony_pony
			namelist = mtrspony_namelists_pony
			traits = {
				trait = trait_pc_continental_preference
				trait = trait_talented
				trait = trait_cutie_marks
				trait = trait_sedentary
			}

			effect = {
				save_event_target_as = pony_species
				set_citizenship_type = { country = ROOT type = citizenship_limited }
				# set_military_service_type = { country = ROOT type = military_service_full }
			}
		}
		create_species = {
			name = "PRESCRIPTED_species_dragon"
			plural = "PRESCRIPTED_species_plural_dragon"
			adjective = "PRESCRIPTED_species_adjective_dragon"
			class = MTRSPONY
			portrait = mtrspony_dragon
			namelist = mtrspony_namelists_dragon
			traits = {
				trait = trait_pc_continental_preference
				trait = trait_strong
				trait = trait_venerable
				trait = trait_slow_breeders
				trait = trait_solitary
			}

			effect = {
				save_event_target_as = dragon_species
				set_citizenship_type = { country = ROOT type = citizenship_limited }
				# set_military_service_type = { country = ROOT type = military_service_full }
			}
		}
		create_species = {
			name = "PRESCRIPTED_species_kirin"
			plural = "PRESCRIPTED_species_plural_kirin"
			adjective = "PRESCRIPTED_species_adjective_kirin"
			class = MTRSPONY
			portrait = mtrspony_kirin
			namelist = mtrspony_namelists_kirin
			traits = {
				trait = trait_pc_continental_preference
				trait = trait_agrarian
				trait = trait_natural_engineers
				trait = trait_wasteful
			}

			effect = {
				save_event_target_as = kirin_species
				set_citizenship_type = { country = ROOT type = citizenship_limited }
				# set_military_service_type = { country = ROOT type = military_service_full }
			}
		}
		create_species = {
			name = "PRESCRIPTED_species_changeling"
			plural = "PRESCRIPTED_species_plural_changeling"
			adjective = "PRESCRIPTED_species_adjective_changeling"
			class = MTRSPONY
			portrait = mtrspony_changeling
			namelist = mtrspony_namelists_changeling
			traits = {
				trait = trait_pc_continental_preference
				trait = trait_communal
				trait = trait_adaptive
				trait = trait_wasteful
			}

			effect = {
				save_event_target_as = changeling_species
				set_citizenship_type = { country = ROOT type = citizenship_limited }
				# set_military_service_type = { country = ROOT type = military_service_full }
			}
		}

		capital_scope = {
			while = {
				count = 15

				random_owned_pop = { kill_pop = yes }
			}
			while = {
				count = 10

				create_pop = {
					species = event_target:pony_species
					ethos = random
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:dragon_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:kirin_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:changeling_species
					ethos = random
				}
			}
		}

		# Remove the auto-generated Grover XI.
		random_owned_leader = {
			limit = {
				NOT = { has_leader_flag = is_created_leader }
				has_trait = trait_ruler_charismatic
			}

			kill_leader = {
				show_notification = no
				fire = yes
			}
		}

		event_target:leader_grover_xi = {
			set_name = "PRESCRIPTED_ruler_name_grover_xi"
		}
	}
}
