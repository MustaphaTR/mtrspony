namespace = gestart

#Setup Convocation of Creatures
country_event = {
	id = gestart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = griffonian_empire
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_grover_xi"
			custom_description = "PRESCRIPTED_ruler_name_grover_xi_desc"
			species = ROOT.species
			gender = male
			class = official
			skill = 1
			set_age = 19
			background_ethic = "ethic_authoritarian"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = trait_imperial_heir
				1 = trait_ruler_charismatic
			}

			effect = {
				save_global_event_target_as = leader_grover_xi
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_grover_xi_predefined"
				ROOT = {
					leader = {
						kill_leader = {
							show_notification = no
							fire = yes
						}
					}
					assign_leader = event_target:leader_grover_xi
				}
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		set_mtrspony_citizenship_limited = yes
		event_target:changeling_species@ROOT = { set_military_service_type = { country = ROOT type = military_service_full } }

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}

		event_target:leader_grover_xi = {
			set_name = "PRESCRIPTED_ruler_name_grover_xi"
		}
	}
}
