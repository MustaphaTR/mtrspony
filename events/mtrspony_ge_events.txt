#Setup Convocation of Creatures
country_event = {
	id = gestart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = griffonian_empire
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_grover_xi"
			custom_description = "PRESCRIPTED_ruler_name_grover_xi_desc"
			species = ROOT.species
			gender = male
			class = official
			skill = 1
			set_age = 19
			background_ethic = ethic_authoritarian
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = trait_imperial_heir
				1 = trait_ruler_charismatic
			}

			effect = {
				save_global_event_target_as = leader_grover_xi
				set_leader_flag = leader_grover_xi
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_grover_xi_predefined"
				ROOT = {
					leader = {
						kill_leader = {
							show_notification = no
							fire = yes
						}
					}
					assign_leader = event_target:leader_grover_xi
				}
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_theresa_eagleclaw"
			custom_description = "PRESCRIPTED_ruler_name_theresa_eagleclaw_desc"
			species = ROOT.species
			gender = female
			class = official
			skill = 1
			set_age = 36
			background_ethic = ethic_authoritarian
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				1 = trait_ruler_fertility_preacher
			}

			effect = {
				save_global_event_target_as = leader_theresa_eagleclaw
				set_leader_flag = leader_theresa_eagleclaw
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_theresa_eagleclaw_predefined"
				ROOT = {
					capital_scope = {
						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = event_target:leader_winter
					}
				}
				set_council_position = councilor_state
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		set_mtrspony_citizenship_limited = yes
		event_target:changeling_species@ROOT = { set_military_service_full = { COUNTRY = ROOT } }
		event_target:diamond_dog_species@ROOT = { set_citizenship_full = { COUNTRY = ROOT } }
		event_target:diamond_dog_species@ROOT = { set_military_service_full = { COUNTRY = ROOT } }

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}
		refresh_leader_pool = yes

		create_leader = {
			name = "PRESCRIPTED_ruler_name_max_von_hundsfeld"
			custom_description = "PRESCRIPTED_ruler_name_max_von_hundsfeld_desc"
			species = event_target:diamond_dog_species@ROOT
			gender = male
			class = commander
			skill = 1
			set_age = 58
			background_ethic = ethic_militarist
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				1 = leader_trait_defence_engineer
			}

			effect = {
				save_global_event_target_as = leader_max_von_hundsfeld
				set_leader_flag = leader_max_von_hundsfeld
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_diamond_dog_max_von_hundsfeld_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_military }

						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_defense
			}
		}

		event_target:leader_grover_xi = {
			set_name = "PRESCRIPTED_ruler_name_grover_xi"
		}

		country_event = { id = gestart.1001 days = 1 }
	}
}

# Properly set the background job and ethics.
country_event = {
	id = gestart.1001
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		event_target:leader_grover_xi = {
			change_background_ethic = ethic_authoritarian
		}
		event_target:leader_theresa_eagleclaw = {
			change_background_ethic = ethic_authoritarian
		}
		event_target:leader_max_von_hundsfeld = {
			change_background_ethic = ethic_militarist
		}
	}
}
