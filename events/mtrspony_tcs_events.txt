#Setup The Crusader State

	id = tcsstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = the_crusader_state
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_sweetie_belle"
			custom_description = "PRESCRIPTED_ruler_name_sweetie_belle_desc"
			species = ROOT.species
			gender = female
			class = scientist
			skill = 1
			set_age = 281
			immortal = yes
			background_job = "head_researcher"
			background_ethic = "ethic_xenophile"
			randomize_traits = no
			traits = {
				trait = leader_trait_expertise_field_manipulation
				trait = trait_ruler_eye_for_talent
			}

			effect = {
				save_global_event_target_as = leader_sweetie_belle
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_unicorn_sweetie_belle_older_predefined"
				ROOT = { assign_leader = event_target:leader_sweetie_belle }
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_apple_bloom"
			custom_description = "PRESCRIPTED_ruler_name_apple_bloom_desc"
			species = ROOT.species
			gender = female
			class = scientist
			skill = 1
			set_age = 281
			immortal = yes
			background_job = "farmer"
			background_ethic = "ethic_egalitarian"
			randomize_traits = no
			traits = {
				trait = leader_trait_expertise_industry
				trait = trait_ruler_eye_for_talent
			}

			effect = {
				save_global_event_target_as = leader_apple_bloom
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_apple_bloom_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_science_ship }

						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_research
			}
		}
		create_leader = {
			name = "PRESCRIPTED_ruler_name_scootaloo"
			custom_description = "PRESCRIPTED_ruler_name_scootaloo_desc"
			species = ROOT.species
			gender = female
			class = commander
			skill = 1
			set_age = 281
			immortal = yes
			background_job = "commanding_officer"
			background_ethic = "ethic_militarist"
			randomize_traits = no
			traits = {
				trait = leader_trait_cyborg
				trait = leader_trait_scout
				trait = trait_ruler_eye_for_talent
			}

			effect = {
				save_global_event_target_as = leader_scootaloo
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_pegasus_scootaloo_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_military }

						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_defense
			}
		}
		create_leader = {
			name = "PRESCRIPTED_ruler_name_babs_seed"
			custom_description = "PRESCRIPTED_ruler_name_babs_seed_desc"
			species = ROOT.species
			gender = female
			class = official
			skill = 1
			set_age = 281
			immortal = yes
			background_job = "hairdresser"
			background_ethic = "ethic_egalitarian"
			randomize_traits = no
			traits = {
				trait = leader_trait_urbanist
				trait = leader_trait_trade_focus
			}

			effect = {
				save_global_event_target_as = leader_babs_seed
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_earth_babs_seed_predefined"
				set_council_position = councilor_state
			}
		}

		create_griffon_species = yes
		event_target:griffon_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_hippogriff_species = yes
		event_target:hippogriff_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_dragon_species = yes
		event_target:dragon_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_kirin_species = yes
		event_target:kirin_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_changeling_species = yes
		event_target:changeling_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_yak_species = yes
		event_target:yak_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		create_zebra_species = yes
		event_target:zebra_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_military_service_type = { country = ROOT type = military_service_full }
		}

		capital_scope = {
			while = {
				count = 18

				random_owned_pop = { kill_pop = yes }
			}
			while = {
				count = 5

				create_pop = {
					species = event_target:griffon_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:griffon_species
					ethos = { ethic = ethic_spiritualist }
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:hippogriff_species
					ethos = random
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:dragon_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:kirin_species
					ethos = random
				}
			}
			while = {
				count = 2

				create_pop = {
					species = event_target:changeling_species
					ethos = random
				}
			}
			while = {
				count = 1

				create_pop = {
					species = event_target:yak_species
					ethos = random
				}
			}
			while = {
				count = 4

				create_pop = {
					species = event_target:zebra_species
					ethos = random
				}
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_gabby"
			custom_description = "PRESCRIPTED_ruler_name_gabby_desc"
			species = event_target:griffon_species
			gender = female
			class = official
			skill = 1
			set_age = 285
			immortal = yes
			background_job = "mailgriffon"
			background_ethic = "ethic_spiritualist"
			randomize_traits = no
			traits = {
				trait = leader_trait_righteous
				trait = leader_trait_utopian_idealist
			}

			effect = {
				save_global_event_target_as = leader_gabby
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_griffon_gabby_predefined"
				ROOT = { capital_scope = { assign_leader = event_target:leader_gabby } }
			}
		}

		# Custom names for starting ships, these are Battleships in the referenced story, but i'm just using them for starting Corvettes.
		random_owned_ship = {
			limit = {
				is_ship_size = corvette
				NOT = { has_ship_flag = has_name_setup }
			}
			set_name = "PRESCRIPTED_ship_name_tcsss_halberd"
			set_ship_flag = has_name_setup
		}
		random_owned_ship = {
			limit = {
				is_ship_size = corvette
				NOT = { has_ship_flag = has_name_setup }
			}
			set_name = "PRESCRIPTED_ship_name_tcsss_warhammer"
			set_ship_flag = has_name_setup
		}
		random_owned_ship = {
			limit = {
				is_ship_size = corvette
				NOT = { has_ship_flag = has_name_setup }
			}
			set_name = "PRESCRIPTED_ship_name_tcsss_unique_horn"
			set_ship_flag = has_name_setup
		}

		# Set policies.
		set_policy = {
			policy = subjugation_war_terms
			option = oppressive_terms
			cooldown = no
		}
		set_policy = {
			policy = orbital_bombardment
			option = orbital_bombardment_selective
			cooldown = no
		}
		# Not available at the start, so can't be set there.
		# set_policy = {
		#	policy = enlightenment
		#	option = enlightenment_allowed
		#	cooldown = no
		# }

		# Remove the genericos.
		every_owned_leader = {
			limit = {
				NOT = { has_leader_flag = is_created_leader }
			}

			kill_leader = {
				show_notification = no
				fire = yes
			}
		}
	}
}
