# Scootaloo starts with Limited Cyborg trait, which means he can't get Psionics. Let her progress with the Cybernetics, even if we are going to another path.
country_event = {
	id = tcsstart.3
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = the_crusader_state
		OR = {
			has_tradition = tr_psionics_adopt
			has_tradition = tr_psionics_shroud_adopt
		}
	}
	immediate = {
		event_target:leader_scootaloo = {
			leader_event = {
				id = tcsstart.4
				days = 1440
				random = 720
			}
		}
	}
}
leader_event = {
	id = tcsstart.4
	title = "tcsstart.4.name"
	desc = "tcsstart.4.desc"
	picture = GFX_evt_cybernetics
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_renowned"

	picture_event_data = {
		portrait = root
	}

	trigger = {
		has_trait = leader_trait_limited_cyborg
		owner = {
			has_country_flag = the_crusader_state
			OR = {
				has_tradition = tr_psionics_adopt
				has_tradition = tr_psionics_shroud_adopt
			}
		}
	}

	option = {
		name = tcsstart.4.a

		hidden_effect = { remove_trait = leader_trait_limited_cyborg }
		add_trait = {
			trait = leader_trait_cyborg
			show_message = no
		}
		owner = { add_resource = { engineering_research = 5000 } }
	}
}

# Cozy Glow offers to run our Penal Colony.
country_event = {
	id = tcsstart.5
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		has_country_flag = the_crusader_state
		last_increased_tech = tech_penal_colonies
	}
	immediate = {
		set_variable = { which = cozy_recruitment_days value = 360 }
		set_variable = { which = cozy_recruitment_random value = 360 }
		if = {
			limit = { count_starbase_buildings = { type = recruitment_office count > 1 } }

			multiply_variable = { which = cozy_recruitment_days value = 0.8 }
			multiply_variable = { which = cozy_recruitment_random value = 0.8 }
		}
		country_event = {
			id = tcsstart.6
			days = var:cozy_recruitment_days
			random = var:cozy_recruitment_random
		}
	}
}
country_event = { # She only offers becoming leader if we have assigned a penal colony, if not continue to check every month until we do.
	id = tcsstart.6
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = the_crusader_state
		has_technology = tech_penal_colonies
	}
	immediate = {
		if = {
			limit = { any_owned_planet = { has_modifier = penal_colony } }

			country_event = { id = tcsstart.7 }
		}
		else = {
			country_event = { id = tcsstart.6 days = 30 }
		}
	}
}

			SPECIES = event_target:pony_species@ROOT

			TIER = leader_tier_renowned

			DESC = PRESCRIPTED_ruler_name_cozy_glow_desc
			CATCH_PHRASE = PRESCRIPTED_ruler_name_cozy_glow_catch_phrase
		}

			remove_all_traits = yes # Removes the generic Cutie Mark trait added by mtrspony.6.

				limit = { species = { has_trait = trait_cutie_marks } } # Just in case trait was removed, even tho it makes no sense to do it as TSC.

				add_trait = { trait = leader_trait_cutie_mark_cozy_glow }

			if = { # These would've been automatically added normally, but i called remove_all_traits above, it is either this, or check for all the cutie mark traits.
				limit = { species = { has_trait = trait_limited_cybernetic } }

				add_trait = { trait = leader_trait_limited_cyborg }
			}
			else_if = {
				limit = { species = { has_trait = trait_cybernetic } }

				add_trait = { trait = leader_trait_cyborg }
			}
			else_if = {
				limit = { species = { has_trait = trait_psionic } }

				add_trait = { trait = leader_trait_psionic }
			}
			if = {
				limit = { species = { has_trait = trait_erudite } }

				add_trait = { trait = leader_trait_erudite }
			}
			add_trait = {
				trait = leader_trait_redeemed_villain

			}
			add_trait = {
				trait = subclass_official_governor
				consume_selection = yes
			}
			add_trait = {
				trait = leader_trait_alicorn_generic
				consume_selection = yes
			}

				trait = trait_ruler_charismatic
				consume_selection = yes
			}
			add_trait = {
				trait = leader_trait_pop_focused
				consume_selection = yes
			}
