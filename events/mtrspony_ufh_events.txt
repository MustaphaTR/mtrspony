namespace = ufhstart

#Setup United Faustian Hives
country_event = {
	id = ufhstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = united_faustian_hives
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_luna"
			custom_description = "PRESCRIPTED_ruler_name_luna_desc"
			species = ROOT.species
			gender = female
			class = official
			skill = 1
			set_age = 1377
			hide_age = no
			background_job = "noble"
			background_ethic = "ethic_gestalt_consciousness"
			skip_background_generation = yes
			immortal = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_cutie_mark_talent
			}

			effect = {
				save_global_event_target_as = leader_luna
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_luna_predefined"
				ROOT = {
					leader = {
						kill_leader = {
							show_notification = no
							fire = yes
						}
					}
					assign_leader = event_target:leader_luna
				}

				if = {
					limit = { ROOT = { has_ethic = ethic_gestalt_consciousness } }

					add_trait = {
						trait = subclass_official_economy_councilor
						show_message = no
					}
					add_trait = {
						trait = leader_trait_ruler_hive_mind
						show_message = no
					}
				}
				else = {
					add_trait = {
						trait = leader_trait_alicorn_generic
						show_message = no
					}
					change_background_ethic = ethic_spiritualist
				}
				add_trait = {
					trait = leader_trait_cutie_mark_luna
					show_message = no
				}
				remove_trait = leader_trait_cutie_mark_talent
				add_trait = {
					trait = leader_trait_principled
					consume_selection = yes
					show_message = no
				}
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_twilight_sparkle"
			custom_description = "PRESCRIPTED_ruler_name_twilight_sparkle_desc"
			species = ROOT.species
			gender = female
			class = scientist
			skill = 1
			set_age = 291
			immortal = yes
			background_job = "head_researcher"
			background_ethic = "ethic_xenophile"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_alicorn_of_magic

				trait = leader_trait_cutie_mark_twilight_sparkle

				trait = leader_trait_neurotic
			}

			effect = {
				save_global_event_target_as = leader_twilight_sparkle
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_twilight_sparkle_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_science_ship }

						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = PREVPREV
					}
				}

				if = {
					limit = { ROOT = { has_ethic = ethic_gestalt_consciousness } }

					add_trait = {
						trait = leader_trait_meticulous
						consume_selection = yes
						show_message = no
					}
					add_trait = {
						trait = leader_trait_mtrspony_disconnected
						show_message = no
					}
				}
				else = {
					add_trait = {
						trait = leader_trait_spark_of_genius
						consume_selection = yes
						show_message = no
					}
					set_council_position = councilor_research
				}
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_mi_amore_cadenza"
			custom_description = "PRESCRIPTED_ruler_name_mi_amore_cadenza_desc"
			species = ROOT.species
			gender = female
			class = official
			skill = 1
			set_age = 315
			immortal = yes
			background_job = "noble"
			background_ethic = "ethic_gestalt_consciousness"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_alicorn_generic

				trait = leader_trait_cutie_mark_mi_amore_cadenza

				1 = leader_trait_venerated
			}

			effect = {
				save_global_event_target_as = leader_cadance
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_cadance_predefined"
				ROOT = {
					capital_scope = {
						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = event_target:leader_cadance
					}
				}
				if = {
					limit = { ROOT = { NOT = { has_ethic = ethic_gestalt_consciousness } } }

					set_council_position = councilor_state
					change_background_ethic = ethic_xenophile
				}
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_flurry_heart"
			custom_description = "PRESCRIPTED_ruler_name_flurry_heart_desc"
			species = ROOT.species
			gender = female
			class = commander
			skill = 1
			set_age = 267
			immortal = yes
			background_job = "noble"
			background_ethic = "ethic_gestalt_consciousness"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_alicorn_generic

				trait = leader_trait_cutie_mark_flurry_heart

				trait = leader_trait_stubborn
			}

			effect = {
				save_global_event_target_as = leader_flurry_heart
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_flurry_heart_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_military }

						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = PREVPREV
					}
				}
				if = {
					limit = { ROOT = { NOT = { has_ethic = ethic_gestalt_consciousness } } }

					add_trait = {
						trait = trait_ruler_logistic_understanding
						consume_selection = yes
						show_message = no
					}
					set_council_position = councilor_defense
					change_background_ethic = ethic_spiritualist
				}
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}
	}
}
