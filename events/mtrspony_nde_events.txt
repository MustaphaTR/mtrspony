			randomize_traits = no
			traits = {
				trait = leader_trait_mtrspony_immortal

				1 = trait_ruler_warlike
			}

			effect = {
				save_global_event_target_as = leader_arcin_the_defiler
				set_leader_flag = is_created_leader

				ROOT = {

			}
		}

				change_leader_portrait = "mtrspony_dragon_charcoal_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_military }

						leader = {
							kill_leader = {
								show_notification = no
								fire = yes

					capital_scope = {
						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = event_target:leader_winter
					}
				}
				set_council_position = councilor_state
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		event_target:pony_species = { set_slavery_type = { country = ROOT type = slavery_normal } }

			set_citizenship_type = { country = ROOT type = citizenship_limited }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:hippogriff_species@ROOT = { set_slavery_type = { country = ROOT type = slavery_normal } }
		event_target:kirin_species@ROOT = { set_slavery_type = { country = ROOT type = slavery_normal } }

			set_slavery_type = { country = ROOT type = slavery_military }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:yak_species@ROOT = {
			set_slavery_type = { country = ROOT type = slavery_military }
			set_military_service_type = { country = ROOT type = military_service_full }

				set_habitability_trait = { trait = trait_pc_arctic_preference }
				change_species_characteristics = { add_trait = trait_survivor }
			}
		}

		capital_scope = {
			remove_deposit = d_failing_infrastructure_faust
			add_deposit = d_failing_infrastructure_frost

			add_original_species_pc_trait_effect = { SPECIES = event_target:pony_species@ROOT TRAIT = trait_pc_continental_preference }
			add_original_species_pc_trait_effect = { SPECIES = event_target:griffon_species@ROOT TRAIT = trait_pc_arctic_preference }

			add_original_species_pc_trait_effect = { SPECIES = event_target:changeling_species@ROOT TRAIT = trait_pc_arctic_preference }
			add_original_species_pc_trait_effect = { SPECIES = event_target:hippogriff_species@ROOT TRAIT = trait_pc_continental_preference }
			add_original_species_pc_trait_effect = { SPECIES = event_target:kirin_species@ROOT TRAIT = trait_pc_tropical_preference }
			add_original_species_pc_trait_effect = { SPECIES = event_target:yak_species@ROOT TRAIT = trait_pc_alpine_preference }

			calculate_mtrspony_pop_count_keep_majority = yes
			set_variable = { which = griffon_pops value = 675 } # Placeholder, most of these should be Penguins.
			set_variable = { which = changeling_pops value = 475 }
			set_variable = { which = yak_pops value = 450 }
			set_variable = { which = polar_bear_pops value = 425 }
			set_variable = { which = pony_pops value = 300 }
			set_variable = { which = zebra_pops value = 250 }
			set_variable = { which = deer_pops value = 250 }
			set_variable = { which = hippogriff_pops value = 150 }
			set_variable = { which = kirin_pops value = 150 }
			if = {
				limit = { 
					count_owned_pop_amount = {
						limit = { is_same_species = owner_main_species }
						count > 5199
					}
				}

				set_variable = { which = griffon_pops value = 760 } # Placeholder, most of these should be Penguins.
				set_variable = { which = changeling_pops value = 535 }
				set_variable = { which = yak_pops value = 505 }
				set_variable = { which = polar_bear_pops value = 480 }
				set_variable = { which = pony_pops value = 335 }
				set_variable = { which = zebra_pops value = 280 }
				set_variable = { which = deer_pops value = 280 }
				set_variable = { which = hippogriff_pops value = 170 }
				set_variable = { which = kirin_pops value = 170 }
			}
			spawn_other_mtrspony_species = { IGNORE_PC = yes }
			spawn_mtrspony_pops = { COUNTRY = owner SPECIES = polar_bear }
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_soeren_of_svartpels"
			custom_description = "PRESCRIPTED_ruler_name_soeren_of_svartpels_desc"
			species = event_target:polar_bear_species@ROOT
			gender = male
			class = scientist
			skill = 1
			set_age = 41
			background_ethic = "ethic_materialist"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				1 = leader_trait_carefree
			}

			effect = {
				save_global_event_target_as = leader_soeren_of_svartpels
				set_leader_flag = is_created_leader
				# change_leader_portrait = "mtrspony_polar_bear_soeren_of_svartpels_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_science_ship }

						leader = {
							kill_leader = {
								show_notification = no
								fire = yes
							}
						}
						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_research
			}
		}
	}
}
