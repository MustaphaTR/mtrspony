namespace = cocstart

#Setup Convocation of Creatures
country_event = {
	id = cocstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = convocation_of_creatures
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_dusk_shine"
			custom_description = "PRESCRIPTED_ruler_name_dusk_shine_desc"
			species = ROOT.species
			gender = male
			class = scientist
			skill = 1
			set_age = 291
			immortal = yes
			background_job = head_researcher
			background_ethic = ethic_xenophile
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_alicorn_of_magic

				trait = leader_trait_cutie_mark_dusk_shine

				1 = leader_trait_spark_of_genius
				trait = leader_trait_neurotic
			}

			effect = {
				save_global_event_target_as = leader_dusk_shine
				set_leader_flag = leader_dusk_shine
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_alicorn_dusk_shine_predefined"
				ROOT = {
					leader = {
						kill_leader = {
							show_notification = no
							fire = yes
						}
					}
					assign_leader = event_target:leader_dusk_shine
				}
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}
		refresh_leader_pool = yes

		country_event = { id = cocstart.1001 days = 1 }
	}
}

# Properly set the background job and ethics.
country_event = {
	id = cocstart.1001
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		event_target:leader_dusk_shine = {
			change_background_job = head_researcher
			change_background_ethic = ethic_xenophile
		}
	}
}
