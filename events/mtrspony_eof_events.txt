namespace = eofstart

#Setup Convocation of Creatures
country_event = {
	id = eofstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = empire_of_friendship
	}
	immediate = {
		if = { # Unfortunately add_trait makes pop ups appear, so have to duplicate the whole thing.
			limit = { has_under_one_rule_origin = yes }

			create_leader = {
				name = "PRESCRIPTED_ruler_name_cozy_glow"
				custom_description = "PRESCRIPTED_ruler_name_cozy_glow_desc"
				species = ROOT.species
				gender = female
				class = official
				skill = 4
				set_age = 48
				background_job = "friendship_student"
				background_ethic = "ethic_authoritarian"
				randomize_traits = no
				traits = {
					trait = leader_trait_politician
					trait = trait_ruler_champion_of_the_people

					trait = leader_trait_legendary_leader
					trait = leader_trait_legendary_gunboat_diplomacy
					trait = leader_trait_legendary_natural_ruler
					trait = leader_trait_legendary_warmonger
				}

				effect = {
					save_global_event_target_as = leader_cozy_glow
					set_leader_flag = is_created_leader
					change_leader_portrait = "mtrspony_pony_pegasus_cozy_glow_older_predefined"
					ROOT = { assign_leader = event_target:leader_cozy_glow }

					set_leader_flag = paragon_origin_legendary_leader
				}
			}
		}
		else = {
			create_leader = {
				name = "PRESCRIPTED_ruler_name_cozy_glow"
				custom_description = "PRESCRIPTED_ruler_name_cozy_glow_desc"
				species = ROOT.species
				gender = female
				class = official
				skill = 4
				set_age = 48
				background_job = "friendship_student"
				background_ethic = "ethic_authoritarian"
				randomize_traits = no
				traits = {
					trait = leader_trait_politician
					trait = trait_ruler_champion_of_the_people

					trait = leader_trait_resilient_2
					trait = trait_ruler_charismatic
				}

				effect = {
					save_global_event_target_as = leader_cozy_glow
					set_leader_flag = is_created_leader
					change_leader_portrait = "mtrspony_pony_pegasus_cozy_glow_older_predefined"
					ROOT = { assign_leader = event_target:leader_cozy_glow }
				}
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes
		species = { set_living_standard = { country = ROOT type = living_standard_good } }
		event_target:griffon_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:hippogriff_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:dragon_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:kirin_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:changeling_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:yak_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}
		event_target:zebra_species = {
			set_citizenship_type = { country = ROOT type = citizenship_full }
			set_living_standard = { country = ROOT type = living_standard_good }
			set_military_service_type = { country = ROOT type = military_service_full }
		}

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}

		# Remove the auto-generated Cozy Glow.
		random_owned_leader = {
			limit = {
				NOT = { has_leader_flag = is_created_leader }
				OR = {
					has_trait = leader_trait_legendary_gunboat_diplomacy
					has_trait = trait_ruler_charismatic
				}
			}

			kill_leader = {
				show_notification = no
				fire = yes
			}
		}
	}
}
