# Properly set the background job and ethics.
country_event = {
	id = dosstart.1001
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		event_target:leader_scootaloo_1064 = {
			change_background_job = commanding_officer
			change_background_ethic = ethic_militarist
		}
	}
}

# Reapply named character potraits, they seem to reset after loading a save or when species got modified.
country_event = {
	id = dosstart.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = department_of_scootascience
	}
	immediate = {
		event_target:leader_scootaloo_1064 = {
			change_leader_portrait = "mtrspony_pony_its_all_scootaloo_n1064_predefined"
		}
	}
}

# Every Scootaloo gets the Scootaloo Cutie Mark.
country_event = {
	id = dosstart.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = department_of_scootascience
		from = {
			species = {
				has_species_flag = is_scootaloo_species
				has_trait = trait_cutie_marks
			}
		}
	}
	immediate = {
		from = {
			remove_cutie_mark_traits = yes
			add_trait = {
				trait = leader_trait_cutie_mark_scootaloo
				show_message = no
			}
		}
	}
}
