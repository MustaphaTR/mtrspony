namespace = fcstart

#Setup Faustian Conglomerate
country_event = {
	id = fcstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = faustian_conglomerate
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_fickle_current"
			custom_description = "PRESCRIPTED_ruler_name_fickle_current_desc"
			species = ROOT.species
			gender = male
			class = official
			skill = 2
			set_age = 56
			background_job = "politician"
			background_ethic = "ethic_materialist"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				1 = leader_trait_principled
				trait = leader_trait_entrepreneur
			}

			effect = {
				save_global_event_target_as = leader_fickle_current
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_kirin_fickle_current_predefined"
				ROOT = {
					leader = {
						kill_leader = {
							show_notification = no
							fire = yes
						}
					}
					assign_leader = event_target:leader_fickle_current
				}
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}

		# Custom names for starting ships.
		random_owned_ship = {
			limit = {
				is_ship_size = corvette
			}
			set_name = "PRESCRIPTED_ship_name_fcs_current"
		}
	}
}

# Reapply named character potraits, they seem to reset after loading a save or when species got modified.
country_event = {
	id = fcstart.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = revolutionary_faust
	}
	immediate = {
		event_target:leader_fickle_current = {
			change_leader_portrait = "mtrspony_kirin_fickle_current_predefined"
		}
	}
}
