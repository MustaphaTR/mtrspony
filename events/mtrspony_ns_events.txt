#Setup Novyy Stalliongrad
country_event = {
	id = nsstart.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = novyy_stalliongrad
	}
	immediate = {
		create_leader = {
			name = "PRESCRIPTED_ruler_name_tall_cherry"
			custom_description = "PRESCRIPTED_ruler_name_tall_cherry_desc"
			species = ROOT.species
			gender = male
			class = commander
			skill = 1
			set_age = 62
			background_job = "commanding_officer"
			background_ethic = "ethic_militarist"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_cutie_mark_statecraft

				1 = trait_ruler_charismatic
			}

			effect = {
				save_global_event_target_as = leader_tall_cherry
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_earth_tall_cherry_predefined"
				ROOT = { assign_leader = event_target:leader_tall_cherry }
			}
		}

		create_leader = {
			name = "PRESCRIPTED_ruler_name_drew_nikolayev"
			custom_description = "PRESCRIPTED_ruler_name_drew_nikolayev_desc"
			species = ROOT.species
			gender = male
			class = commander
			skill = 1
			set_age = 52
			background_job = "commanding_officer"
			background_ethic = "ethic_militarist"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_cutie_mark_particles

				1 = leader_trait_aggressive
				trait = leader_trait_reckless
			}

			effect = {
				save_global_event_target_as = leader_drew_nikolayev
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_pegasus_drew_nikolayev_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_military }

						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_defense
			}
		}
		create_leader = {
			name = "PRESCRIPTED_ruler_name_petra_stormmare"
			custom_description = "PRESCRIPTED_ruler_name_petra_stormmare_desc"
			species = ROOT.species
			gender = female
			class = scientist
			skill = 1
			set_age = 54
			background_job = "head_researcher"
			background_ethic = "ethic_materialist"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_cutie_mark_field_manupilation

				1 = leader_trait_adaptable
			}

			effect = {
				save_global_event_target_as = leader_petra_stormmare
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_pegasus_petra_stormmare_predefined"
				ROOT = {
					random_owned_fleet = {
						limit = { is_ship_class = shipclass_science_ship }

						assign_leader = PREVPREV
					}
				}
				set_council_position = councilor_research
			}
		}
		create_leader = {
			name = "PRESCRIPTED_ruler_name_dashing_gift"
			custom_description = "PRESCRIPTED_ruler_name_dashing_gift_desc"
			species = ROOT.species
			gender = female
			class = official
			skill = 1
			set_age = 34
			background_job = "intelligence_officer"
			background_ethic = "ethic_egalitarian"
			skip_background_generation = yes
			randomize_traits = no
			traits = {
				trait = leader_trait_cutie_mark_computing

				1 = leader_trait_army_veteran
			}

			effect = {
				save_global_event_target_as = leader_dashing_gift
				set_leader_flag = is_created_leader
				change_leader_portrait = "mtrspony_pony_pegasus_dashing_gift_predefined"
				ROOT = { capital_scope = { assign_leader = event_target:leader_dashing_gift } }
				set_council_position = councilor_state
			}
		}

		save_current_mtrspony_species = yes
		create_other_mtrspony_species = yes

		capital_scope = {
			calculate_mtrspony_pop_count_keep_majority = yes
			spawn_other_mtrspony_species = yes
		}
		if = {
			limit = { has_origin = origin_lost_colony }

			species = {
				home_planet = {
					owner = {
						every_owned_planet = {
							calculate_mtrspony_pop_count_default = yes
							spawn_other_mtrspony_species = yes
						}
					}
				}
			}
		}

		random_planet_within_border = {
			limit = {
				is_planet_class = pc_frozen
				planet_size = 5
			}

			set_name = "NAME_New_Selena"
		}
		random_planet_within_border = {
			limit = {
				is_planet_class = pc_frozen
				planet_size = 8
			}

			set_name = "NAME_Eos"
		}
		random_planet_within_border = {
			limit = {
				is_planet_class = pc_frozen
				planet_size = 20
			}

			set_name = "NAME_Yacks_Planet"
		}

		# Remove the genericos.
		every_owned_leader = {
			limit = {
				NOT = { has_leader_flag = is_created_leader }
			}

			kill_leader = {
				show_notification = no
				fire = yes
			}
		}
	}
}
